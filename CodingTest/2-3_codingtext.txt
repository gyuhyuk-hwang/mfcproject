#include <string>
#include <vector>
#include <set>

using namespace std;

int solution(vector<vector<int>> dots) {
    int minX = dots[0][0];
    int maxX = dots[0][0];
    int minY = dots[0][1];
    int maxY = dots[0][1];
    
    for(int i=1; i<dots.size(); i++){
        if(dots[i][0] <= minX){
            minX = dots[i][0];
        }
        else{
            maxX = dots[i][0];
        }
        
        if(dots[i][1] <= minY){
            minY = dots[i][1];
        }
        else{
            maxY = dots[i][1];
        }
        
    }
    
    
    
    return (maxX-minX)*(maxY-minY);
}
